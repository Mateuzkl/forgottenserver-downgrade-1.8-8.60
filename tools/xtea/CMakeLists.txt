cmake_minimum_required(VERSION 3.20)
project(tfs_xtea_research LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(TFS_XTEA_FORCE_SCALAR "Force the baseline backend" OFF)
option(ENABLE_NATIVE_OPTIMIZATIONS "Match native Release benchmark flags" OFF)
option(ENABLE_ASAN "Instrument focused tests with ASan/UBSan" OFF)
option(ENABLE_IPO "Enable Release IPO when supported" ON)
include("../../cmake/Xtea.cmake")
if(ENABLE_IPO AND NOT ENABLE_ASAN)
    include(CheckIPOSupported)
    check_ipo_supported(RESULT supported)
    if(supported)
        set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
    endif()
endif()
add_library(xtea STATIC ../../src/xtea.cpp ../../src/xtea_sse2.cpp ../../src/xtea_avx2.cpp)
tfs_configure_xtea(xtea)
if(MSVC)
    target_compile_options(xtea PRIVATE /W4)
else()
    target_compile_options(xtea PRIVATE -Wall -Wextra -Werror -pedantic)
    if(ENABLE_NATIVE_OPTIMIZATIONS)
        add_compile_options(-march=native -mtune=native)
        target_compile_options(xtea PRIVATE -march=native -mtune=native)
    endif()
endif()
if(ENABLE_ASAN AND NOT MSVC)
    target_compile_options(xtea PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(xtea PUBLIC -fsanitize=address,undefined)
endif()
find_package(Threads REQUIRED)
add_executable(test_xtea_simd ../../src/tests/test_xtea_simd.cpp)
target_link_libraries(test_xtea_simd PRIVATE xtea Threads::Threads)
if(TFS_XTEA_FORCE_SCALAR)
    target_compile_definitions(test_xtea_simd PRIVATE TFS_XTEA_FORCE_SCALAR=1)
endif()
add_executable(bench_xtea benchmark.cpp)
target_link_libraries(bench_xtea PRIVATE xtea)
enable_testing()
add_test(NAME test_xtea_simd COMMAND test_xtea_simd)
